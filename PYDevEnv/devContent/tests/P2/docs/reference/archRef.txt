def build_model(hp):
    model = Sequential()
    model.add(Input(shape=(window_size,)))

    # Tune number of hidden layers (1 to 4)
    num_layers = hp.Int('num_layers', 1, 4)

    for i in range(num_layers):
        units = hp.Int(f'units_{i}', min_value=32, max_value=256, step=10)
        activation = hp.Choice(f'activation_{i}', values=['relu', 'tanh'])

        # If LeakyReLU is selected, add it as a layer instead of a string
        if activation == "relu":
            model.add(Dense(units, activation='relu'))
        elif activation == "tanh":
            model.add(Dense(units, activation='tanh'))
        else:  # Use LeakyReLU as a separate layer
            model.add(Dense(units))
            model.add(LeakyReLU())  # Explicitly add LeakyReLU layer

    model.add(Dense(1))  # Output layer

    # Tune optimizer learning rate
    hp_learning_rate = hp.Choice('learning_rate', values=[1e-2, 1e-3, 1e-4])
    optimizer = Adam(learning_rate=hp_learning_rate)

    model.compile(optimizer=optimizer, loss='mse', metrics=['mae'])

    return model


class MyTuner(kt.Tuner):
    def __init__(self, *args, **kwargs):
        super(MyTuner, self).__init__(*args, **kwargs)
        self.aux = []

    def run_trial(self, trial, *args, **kwargs):
        hp = trial.hyperparameters
        model = self.hypermodel.build(hp)

        batch_size = hp.Choice('batch_size', values=batch_sizes)
        epoch = epochs

        model.fit(kwargs['X_train'], kwargs['y_train'],
                  validation_data=(kwargs['X_test'], kwargs['y_test']),
                  batch_size=batch_size,
                  epochs=epoch,
                  verbose=1)

        val_loss, val_mae = model.evaluate(kwargs['X_test'], kwargs['y_test'], verbose=1)
        self.aux.append(val_mae)
        self.oracle.update_trial(trial.trial_id, {'val_loss': val_loss, 'val_mae': val_mae})

        self.save_model(trial.trial_id, model)

    def save_model(self, trial_id, model, overwrite=True):
        model_dir = f"{self.directory}/{self.project_name}/trial_{trial_id}"
        os.makedirs(model_dir, exist_ok=True)
        model_path = os.path.join(model_dir, 'model.keras')
        model.save(model_path)

    def load_model(self, trial):
        model_dir = f"{self.directory}/{self.project_name}/trial_{trial.trial_id}"
        model_path = os.path.join(model_dir, 'model.keras')

        if not os.path.exists(model_path):
            raise FileNotFoundError(f"No model file found at {model_path}")

        return tf.keras.models.load_model(model_path)
